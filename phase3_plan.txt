---------------------------------------------------------------------
 1. THE IDEA (SIMPLY)
---------------------------------------------------------------------

Right now ZYRO (the AI model) can only talk. Phase 3 lets it ASK for
small favors -- but it can never DO anything itself.

Think of a hotel guest calling room service: the guest (the model)
can only order from the printed menu. The kitchen (our code) checks
every order against the menu and only then cooks. If the guest orders
something weird, the kitchen just says "sorry, not on the menu" --
no crash, no execution, no danger.

Flow for every user message:

  1. Model replies.
  2. We check the reply: does it contain a JSON tool request?
     Format the model must use (a FIXED menu, hand-written by us):
         {"tool": "list_files", "path": "memory"}
         {"tool": "read_file",  "path": "notes.txt"}
         {"tool": "get_time"}
  3. permissions.py validates the request:
       - tool name must be on the whitelist (only those 3)
       - path must land inside D:\zyro (blocks "..", C:\..., etc.)
       - broken JSON or unknown tool -> a polite refusal message is
         sent back to the model. No crash, ever.
  4. tools.py runs the tool. All tools are READ-ONLY and safe:
       - list_files : names of files in a folder
       - read_file  : first 2000 characters of a text file
       - get_time   : current date and time
  5. The result goes back to the model AS A USER MESSAGE, so it can
     answer in normal words.
  6. Maximum 3 tool rounds per user message -> an endless loop is
     impossible.

import json
from pathlib import Path

# The only folder ZYRO may ever touch.
BASE_DIR = Path(r"D:\zyro")

# The fixed menu of tools. Phase 4 may extend this BY HAND --
# the model can never add anything to it.
ALLOWED_TOOLS = {"list_files", "read_file", "get_time"}

# Which of those tools need a "path" argument.
TOOLS_WITH_PATH = {"list_files", "read_file"}


def _extract_json(text):
    """Find the first {...} block in the model's reply and parse it.
    Returns a dict, or None if the reply holds no JSON object."""
    start = text.find("{")
    end = text.rfind("}")
    if start == -1 or end <= start:
        return None
    try:
        data = json.loads(text[start:end + 1])
    except ValueError:  # not valid JSON
        return None
    return data if isinstance(data, dict) else None


def _resolve_inside(path_text):
    """Turn a requested path into a real full path -- but only if it
    lands inside D:\zyro. Returns the full Path, or None if it would
    land outside. This blocks tricks like '..' or absolute paths
    such as 'C:\\Users\\...'."""
    try:
        full = (BASE_DIR / path_text).resolve()
    except (OSError, ValueError):
        return None
    base = str(BASE_DIR.resolve()).lower()
    full_text = str(full).lower()
    if full_text == base or full_text.startswith(base + "\\"):
        return full
    return None


def check_reply(text):
    """Inspect one model reply (already think-stripped).
    Returns (kind, request, refusal):
      ("none", None, None)    no tool request -- the reply is just chat
      ("bad",  None, msg)     looked like a request but was refused
      ("ok",   request, None) safe to run; request = {"tool": ..., "path": ...}
    """
    data = _extract_json(text)
    if data is None:
        # No parseable JSON. If the reply at least TRIED to name a tool,
        # tell the model its JSON was broken; otherwise it is just chat.
        if '"tool"' in text or "'tool'" in text:
            return ("bad", None,
                    "That tool request was not valid JSON. Reply with exactly: "
                    '{"tool": "read_file", "path": "notes.txt"}')
        return "none", None, None
    if "tool" not in data:
        return "none", None, None  # JSON, but not a tool request

    tool = data.get("tool")
    # isinstance first: a non-string "tool" (e.g. a JSON list) is unhashable
    # and would crash set membership -- the gate must refuse it, not crash.
    if not isinstance(tool, str) or tool not in ALLOWED_TOOLS:
        allowed = ", ".join(sorted(ALLOWED_TOOLS))
        return ("bad", None,
                f"'{tool}' is not an allowed tool. Allowed tools: {allowed}.")

    if tool == "get_time":
        return "ok", {"tool": tool}, None

    path = data.get("path")
    if not isinstance(path, str) or not path.strip():
        if tool == "list_files":
            path = "."  # empty path = the main D:\zyro folder
        else:
            return ("bad", None,
                    "read_file needs a 'path' to a file inside D:\\zyro.")

    full = _resolve_inside(path)
    if full is None:
        return ("bad", None,
                "That path is outside D:\\zyro, so I am not allowed to touch it.")

    return "ok", {"tool": tool, "path": str(full)}, None

import os
from datetime import datetime

# read_file never returns more than this many characters, so a huge
# file cannot blow up the model's small context window.
MAX_READ_CHARS = 2000


def run_tool(request):
    """Run one validated tool request and return the result as a string."""
    tool = request["tool"]
    if tool == "list_files":
        return _list_files(request["path"])
    if tool == "read_file":
        return _read_file(request["path"])
    if tool == "get_time":
        return _get_time()
    return "ERROR: unknown tool."  # can't happen, but stay safe


def _list_files(folder):
    """Return the names in a folder, one per line, sorted."""
    try:
        names = sorted(os.listdir(folder))
    except OSError as e:
        return f"ERROR: cannot list that folder ({e})."
    if not names:
        return "(the folder is empty)"
    return "\n".join(names)


def _read_file(path):
    """Return the first 2000 characters of a text file."""
    try:
        with open(path, "r", encoding="utf-8", errors="replace") as f:
            content = f.read(MAX_READ_CHARS)
    except OSError as e:
        return f"ERROR: cannot read that file ({e})."
    if len(content) == MAX_READ_CHARS:
        content += "\n...[file continues -- first 2000 characters shown]"
    return content if content.strip() else "(the file is empty)"


def _get_time():
    """Return the current local date and time."""
    return datetime.now().strftime("%A %d %B %Y, %H:%M")

import glob
import json
import os
import re
from datetime import datetime

from ollama import chat

from core import permissions, tools  # Phase 3: permission gate + tools

MODEL = "qwen3:1.7b"

# Folder where past chats are saved. It sits next to main.py,
# so it is always D:\zyro\memory no matter where you start the script.
MEMORY_DIR = os.path.join(os.path.dirname(os.path.abspath(__file__)), "memory")

# How many old messages ZYRO re-reads when you continue a session.
# More = better memory but slower replies, so we keep this small.
MAX_LOADED = 10

# Phase 3: how many tool requests ZYRO may act on per user message.
# This makes an endless request loop impossible.
MAX_TOOL_ROUNDS = 3

# ZYRO's short-term memory for this session. It starts with a system
# message (ZYRO's job description) and grows as the conversation goes.
messages = [
    {
        "role": "system",
        "content": "You are ZYRO, a friendly offline assistant on the "
                   "user's Windows 11 PC. Keep answers short, clear and simple.\n"
                   "\n"
                   "You can ask ZYRO's helper to fetch simple facts for you. "
                   "When you need one, reply with ONLY a JSON object, nothing "
                   "else, in one of these exact forms:\n"
                   '{"tool": "list_files", "path": "memory"}\n'
                   '{"tool": "read_file", "path": "notes.txt"}\n'
                   '{"tool": "get_time"}\n'
                   "Paths must be files or folders inside D:\\zyro.\n"
                   'Example: user asks "what files are in the memory folder?" '
                   '-> {"tool": "list_files", "path": "memory"}\n'
                   "After you receive a tool result, answer the user in normal "
                   "words. If no tool is needed, just answer normally.",
    }
]


def save_session():
    """Save the whole conversation, e.g. memory/chat_2026-10-12_18-05.json."""
    if len(messages) <= 1:  # only the system message = nothing was chatted
        return
    os.makedirs(MEMORY_DIR, exist_ok=True)  # no error if folder already exists
    filename = datetime.now().strftime("chat_%Y-%m-%d_%H-%M.json")
    # Safety checks so a save can never land outside the memory folder:
    # 1) keep only the bare file name (strips any folder parts)
    filename = os.path.basename(filename)
    # 2) the name must be exactly chat_YYYY-MM-DD_HH-MM.json (no / \ or ..)
    if not re.fullmatch(r"chat_\d{4}-\d{2}-\d{2}_\d{2}-\d{2}\.json", filename):
        raise ValueError("unexpected session filename: " + filename)
    # 3) after cleaning up the path, it must still sit inside MEMORY_DIR
    path = os.path.normpath(os.path.join(MEMORY_DIR, filename))
    if not path.startswith(os.path.abspath(MEMORY_DIR) + os.sep):
        raise ValueError("refusing to save outside the memory folder")
    with open(path, "w", encoding="utf-8") as f:
        json.dump(messages, f, ensure_ascii=False, indent=2)
    print("Session saved to", path)


print("ZYRO is ready (model:", MODEL + "). Type 'exit' to quit.")

# --- Phase 2: offer to continue the last saved chat ---
sessions = sorted(glob.glob(os.path.join(MEMORY_DIR, "chat_*.json")))
if sessions:
    newest = sessions[-1]  # names contain the date, so sorting puts newest last
    print(f"Found {len(sessions)} saved session(s). Newest: {os.path.basename(newest)}")
    if input("Continue last session? (y/N): ").strip().lower() == "y":
        try:
            with open(newest, "r", encoding="utf-8") as f:
                past = json.load(f)
        except (OSError, json.JSONDecodeError):
            past = None
            print("Could not read that file -- starting fresh instead.")
        if past:
            # Skip the saved system message (we already have a fresh one)
            # and keep only the last MAX_LOADED messages to stay fast.
            past = [m for m in past if m.get("role") != "system"][-MAX_LOADED:]
            messages.extend(past)
            print(f"ZYRO remembered the last {len(past)} messages.")

while True:
    try:
        user_input = input("\nYou: ").strip()
    except (KeyboardInterrupt, EOFError):
        save_session()
        print("\nZYRO: Bye!")
        break

    if user_input.lower() == "exit":
        save_session()
        print("ZYRO: Bye!")
        break
    if not user_input:
        continue  # ignore empty lines

    # Remember what the user just said.
    messages.append({"role": "user", "content": user_input})

    # Phase 3: one user message can trigger a few tool rounds:
    # model asks for a tool -> we run it -> model answers (maybe asks again).
    # permissions.py decides what is allowed; MAX_TOOL_ROUNDS guarantees
    # this can never loop forever.
    for round_num in range(MAX_TOOL_ROUNDS + 1):
        response = chat(model=MODEL, messages=messages)
        reply = response.message.content

        # Qwen3 drafts its reasoning in a <think>...</think> block before the
        # real answer. Delete it (even if it spans many lines) so we only
        # ever see the answer itself.
        reply = re.sub(r"<think>.*?</think>", "", reply, flags=re.DOTALL).strip()
        if not reply:
            reply = "(no visible answer -- ask again)"

        # Remember what ZYRO said, so the next reply has context.
        messages.append({"role": "assistant", "content": reply})
        print("ZYRO:", reply)

        # Is this reply a tool request? The permission gate decides.
        kind, request, refusal = permissions.check_reply(reply)

        if kind == "none":
            break  # a normal chat answer -- nothing to run

        if round_num == MAX_TOOL_ROUNDS:
            print("  [tool limit reached -- request ignored]")
            break

        if kind == "bad":
            print("  [permission refused]")
            feedback = ("[Permission refused] " + refusal +
                        " Either fix the tool request or just answer the user normally.")
        else:  # kind == "ok" -- already validated, safe to run
            print(f"  [tool: {request['tool']}]")
            result = tools.run_tool(request)
            print("  [result sent back to ZYRO]")
            feedback = (f"[Tool result for {request['tool']}]:\n{result}\n\n"
                        "Use this to answer the user's last message in normal words.")

        # The result (or the refusal) goes to the model as a user message,
        # then the loop lets it speak again.
        messages.append({"role": "user", "content": feedback})

 6. RULES THIS DESIGN ENFORCES
---------------------------------------------------------------------

- The model can NEVER run commands. It can only print a JSON order
  from a fixed, hand-written menu. It can never add to the menu.
- Every path is checked to stay inside D:\zyro. "..", absolute paths
  to other folders/drives, and broken paths are all refused politely.
- Invalid JSON or an unknown tool = polite refusal message back to
  the model. The program never crashes on model output.
- Maximum 3 tool rounds per user message -> no endless loops.
- All 3 tools are READ-ONLY. No writing, deleting, or shell.
- Only Python built-ins + the existing "ollama" package. Nothing new
  is installed. Nothing outside D:\zyro is touched.
